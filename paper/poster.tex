\useblockstyle{IntroBox}
\block{Introduction}{
  \useinnerblockstyle{SubSection}

  \innerblock{The Problem}{
    You fit a regression on the full dataset. $R^2$ is fine, p-values are small. A reviewer asks:
    \textit{``Does this hold for each subgroup?''}
    \begin{center}\textbf{You check. It doesn't. Now what?}\end{center}
  }
  \vspace{40pt}
  \innerblock{Simpson's Paradox}{
    \begin{center}\includegraphics[width=0.58\linewidth]{simpson.jpg}\end{center}
    Within each group the trend is positive. Pooled, it reverses. The pooled regression hides subgroup structure.
  }
  \vspace{40pt}
  \innerblock{``It's an Artifact''}{
    \begin{center}
      \begin{tikzpicture}[every node/.style={align=center}, scale=0.95]
        \node[draw=PrimaryBlue!30, fill=PrimaryBlue!5, rounded corners=3pt,
              inner sep=4pt, minimum width=0.16\linewidth] (A) {``The coefs differ\\across groups!''};
        \node[draw=PrimaryBlue!30, fill=PrimaryBlue!5, rounded corners=3pt,
              inner sep=4pt, minimum width=0.16\linewidth, below=1.5cm of A] (B) {``Genuine heterogeneity.\\Model doesn't transport.''};
        \node[draw=AccentTeal!70, fill=AccentTeal!8, rounded corners=3pt,
              inner sep=4pt, minimum width=0.16\linewidth, below=1.5cm of B] (C)
              {\color{AccentTeal}``Wait --- different $X$\\distributions cause\\most of the drift.''};
        \draw[->, PrimaryBlue!40, line width=1pt] (A.south) -- (B.north);
        \draw[->, PrimaryBlue!40, line width=1pt] (B.south) -- (C.north);
      \end{tikzpicture}
    \end{center}
    \vspace{20pt}  
    Coef differences = \textbf{genuine heterogeneity} $+$ \textbf{mechanical artifact} (covariate distribution differences). Standard practice conflates them.
  }
  \vspace{40pt}
  \innerblock{Real-Data: NHANES Blood Pressure}{
    $n = 9{,}254$, systolic BP, age-decade subgroups.
    \begin{center}\includegraphics[width=0.45\linewidth]{../results/nhanes_age_decade/bpsys_coef_by_age_decade_ci.png}\end{center}
    \textbf{SSR = 0.57} \quad|\quad \textbf{Drift: 32.07 $\to$ 7.49} (context-norm.)
    
    \begin{itemize}[leftmargin=*, itemsep=1pt]
      \item Effect \textit{reverses sign} in older groups
      \item Pooled model underperforms in \textbf{every} age decade
      \item $\widetilde{D}_j$ stays large after correction
    \end{itemize}
  }

  \vspace{20pt}
  \innerblock{Five Stability Metrics}{
    \begin{tabular}{@{}p{0.10\linewidth} p{0.48\linewidth} p{0.34\linewidth}@{}}
      \toprule
      \textbf{Metric} & \textbf{Definition} & \textbf{What it captures} \\
      \midrule
      SSR
      & $\mathrm{SSR}_j = \frac{1}{M}\!\sum\!\mathbf{1}(\mathrm{sgn}\,\hat\beta_{j,a}=\mathrm{sgn}\,\hat\beta_{j,b})$
      & Do signs ($+$/$-$) reverse across groups? \\[15pt]
      Std.\ drift
      & $D_j^\star = \frac{\max_g|\hat\beta_{j,g} - \hat\beta_{j,\text{pooled}}|}{\widehat{\text{SE}}(\hat\beta_{j,\text{pooled}})}$
      & How far do subgroup coefs move from pooled? \\[15pt]
      Context drift
      & $\widetilde{D}_j = \dfrac{D_j^\star}{1 + M_j}$,\quad $M_j =$ avg.\ $|\bar{x}_{j,a} - \bar{x}_{j,b}| / s_{j}$
      & Drift corrected for $X$ distribution differences. \\[15pt]
      TPG
      & $\mathrm{TPG}_{a\to b} = \mathrm{RMSE}_{\text{pooled}\to b} - \mathrm{RMSE}_{b\to b}$
      & Prediction loss across groups. \\[15pt]
      CSS
      & $\mathrm{CSS}_{a,b} = \frac{1}{p}\!\sum_{j}\!\frac{|\bar{x}_{j,a} - \bar{x}_{j,b}|}{s_{j}}$
      & How different are $X$ distributions? \\
      \bottomrule
    \end{tabular}
  }

\useblockstyle{IntroBox}
\block{Why it Happens: The Explanation}{

  \useinnerblockstyle{SubSection}

  \innerblock{Key Idea: $D_j^\star = \widetilde{D}_j \times (1 + M_j)$}{
    Different $X$ distributions $\Rightarrow$ different design matrices $\Rightarrow$ mechanical drift even with identical $\beta$.

    \begin{center}
      \colorbox{PrimaryBlue!5}{$\displaystyle D_j^\star \;=\; \widetilde{D}_j \;\times\; (1 + M_j)$}\\[3pt]
      {\small Total = Genuine $\times$ Mechanical}
    \end{center}

    \begin{center}
      $M_j\!\approx\!0 \;\Rightarrow\; \widetilde{D}_j \!\approx\! D_j^\star$ (real)\\
      $M_j\!\uparrow$, $\widetilde{D}_j\!\downarrow \;\Rightarrow$ artifact\\
      $M_j\!\uparrow$, $\widetilde{D}_j\!\uparrow \;\Rightarrow$ genuine
    \end{center}
  }

  \vspace{40pt}
  \innerblock{S1 vs S4: What Differed?}{
    \begin{center}
      \includegraphics[width=0.9\linewidth]{../results/simulation/poster_s1vss4_drift.png}
    \end{center}
    \textbf{S4 breaks:} coefficient shifts cause sign reversals (SSR $<1$); covariate shifts add mechanical drift. Context strips the mechanical part but $\beta$ differences remain.\\[2pt]
    \textbf{S1 stable:} no true heterogeneity $\Rightarrow$ all drift mechanical $\Rightarrow$ $\widetilde{D}_j \to 0$.
  }

  \vspace{20pt}
  \innerblock{What Drives Instability?}{
    \vspace{40pt}
    \centering
    \begin{tikzpicture}[x=\linewidth, y=1cm, every node/.style={align=center}]
      % Top row: five causes
      \node[draw=PrimaryBlue!60, fill=PrimaryBlue!5, rounded corners=3pt,
            inner sep=4pt, minimum width=2.2cm] (G) at (0.10,0) {$G \uparrow$};
      \node[draw=PrimaryBlue!60, fill=PrimaryBlue!5, rounded corners=3pt,
            inner sep=4pt, minimum width=2.2cm] (N) at (0.30,1) {$n_g \downarrow$};
      \node[draw=AccentAmber!80, fill=AccentAmber!5, rounded corners=3pt,
            inner sep=4pt, minimum width=2.2cm] (IMB) at (0.50,1) {Imbalance};
      \node[draw=AlertRed!80, fill=AlertRed!5, rounded corners=3pt,
            inner sep=4pt, minimum width=2.2cm] (HET) at (0.70,1) {Coef.\ het.};
      \node[draw=AccentTeal!80, fill=AccentTeal!5, rounded corners=3pt,
            inner sep=4pt, minimum width=2.2cm] (COV) at (0.90,0) {Cov.\ shift};

      % Bottom row: result
      \node[draw=AlertRed!70, fill=AlertRed!8, rounded corners=5pt,
            inner sep=8pt, minimum width=8cm] (INST) at (0.50,-3.0)
            {\textbf{\Large Regression instability}\\[4pt]\large drift$\uparrow$ \quad SSR$\downarrow$ \quad TPG$\uparrow$};

      % Arrows: all five point down to instability
      \draw[->, thick, PrimaryBlue!50] (G.south) -- (INST.west);
      \draw[->, thick, PrimaryBlue!50] (N.south) -- (INST.north -| 0.30,0);
      \draw[->, thick, AccentAmber!50] (IMB.south) -- (INST.north -| 0.50,0);
      \draw[->, thick, AlertRed!50] (HET.south) -- (INST.north -| 0.70,0);
      \draw[->, thick, AccentTeal!50] (COV.south) -- (INST.east);
    \end{tikzpicture}
    \vspace{30pt}
    \begin{itemize}[leftmargin=*, itemsep=2pt]
      \item These factors interact multiplicatively.
      \item Always check per-group $n$ and imbalance before interpreting subgroup estimates.
    \end{itemize}
    \vspace{20pt}
  }
